function Get-A11V10BytesSha256 {
    param([Parameter(Mandatory)] [byte[]] $Bytes)
    $Algorithm = [Security.Cryptography.SHA256]::Create()
    try {
        [BitConverter]::ToString($Algorithm.ComputeHash($Bytes)).Replace('-', '').ToLowerInvariant()
    }
    finally {
        $Algorithm.Dispose()
    }
}

function Get-A11V10FileSha256 {
    param([Parameter(Mandatory)] [string] $Path)
    $Stream = [IO.File]::Open($Path, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::Read)
    $Algorithm = [Security.Cryptography.SHA256]::Create()
    try {
        [BitConverter]::ToString($Algorithm.ComputeHash($Stream)).Replace('-', '').ToLowerInvariant()
    }
    finally {
        $Algorithm.Dispose()
        $Stream.Dispose()
    }
}

function Assert-A11V10ParentChainOrdinary {
    param([Parameter(Mandatory)] [string] $Path)
    $Cursor = [IO.Path]::GetDirectoryName([IO.Path]::GetFullPath($Path))
    while (-not [string]::IsNullOrEmpty($Cursor)) {
        $Item = Get-Item -LiteralPath $Cursor -Force -ErrorAction Stop
        if (($Item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0 -or -not [string]::IsNullOrEmpty([string]$Item.LinkType)) { throw "linked parent rejected: $Cursor" }
        $Parent = [IO.Directory]::GetParent($Cursor)
        if ($null -eq $Parent) { break }
        $Cursor = $Parent.FullName
    }
}

function Assert-A11V10FileIdentity {
    param(
        [Parameter(Mandatory)] [string] $Path,
        [Parameter(Mandatory)] [long] $ByteCount,
        [Parameter(Mandatory)] [string] $Sha256,
        [Parameter(Mandatory)] [string] $LeafLinkType,
        [string] $FileVersion = '',
        [string] $ProductVersion = ''
    )
    if (-not [IO.Path]::IsPathFullyQualified($Path) -or -not [StringComparer]::OrdinalIgnoreCase.Equals([IO.Path]::GetFullPath($Path), $Path)) { throw "file path rejected: $Path" }
    $Item = Get-Item -LiteralPath $Path -Force -ErrorAction Stop
    if ($Item.PSIsContainer -or [long]$Item.Length -ne $ByteCount) { throw "file byte count rejected: $Path" }
    if (($Item.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0) { throw "file reparse point rejected: $Path" }
    $ActualLink = if ([string]::IsNullOrEmpty([string]$Item.LinkType)) { 'None' } else { [string]$Item.LinkType }
    if (-not [StringComparer]::Ordinal.Equals($ActualLink, $LeafLinkType)) { throw "file link type rejected: $Path" }
    Assert-A11V10ParentChainOrdinary -Path $Path
    if (-not [StringComparer]::Ordinal.Equals((Get-A11V10FileSha256 -Path $Path), $Sha256)) { throw "file digest rejected: $Path" }
    if (-not [string]::IsNullOrEmpty($FileVersion) -and -not [StringComparer]::Ordinal.Equals([string]$Item.VersionInfo.FileVersion, $FileVersion)) { throw "file version rejected: $Path" }
    if (-not [string]::IsNullOrEmpty($ProductVersion) -and -not [StringComparer]::Ordinal.Equals([string]$Item.VersionInfo.ProductVersion, $ProductVersion)) { throw "product version rejected: $Path" }
    return $Item
}

function Assert-A11V10SourceIdentity {
    param(
        [Parameter(Mandatory)] [string] $Path,
        [Parameter(Mandatory)] [long] $ByteCount,
        [Parameter(Mandatory)] [string] $Sha256
    )
    [void](Assert-A11V10FileIdentity -Path $Path -ByteCount $ByteCount -Sha256 $Sha256 -LeafLinkType 'None')
    $Bytes = [IO.File]::ReadAllBytes($Path)
    if ($Bytes.Length -eq 0 -or ($Bytes.Length -ge 3 -and $Bytes[0] -eq 0xEF -and $Bytes[1] -eq 0xBB -and $Bytes[2] -eq 0xBF)) { throw "source BOM/empty rejected: $Path" }
    if ($Bytes[-1] -ne 0x0A -or ($Bytes.Length -gt 1 -and $Bytes[-2] -eq 0x0A) -or $Bytes -contains 0x0D) { throw "source line ending rejected: $Path" }
    $RoundTrip = $Utf8.GetBytes($Utf8.GetString($Bytes))
    if ($RoundTrip.Length -ne $Bytes.Length -or -not [StringComparer]::Ordinal.Equals((Get-A11V10BytesSha256 -Bytes $RoundTrip), $Sha256)) { throw "source UTF-8 round trip rejected: $Path" }
    $Streams = @(Get-Item -LiteralPath $Path -Stream * -ErrorAction Stop)
    if ($Streams.Count -ne 1 -or [string]$Streams[0].Stream -cne ':$DATA') { throw "source stream rejected: $Path" }
}
